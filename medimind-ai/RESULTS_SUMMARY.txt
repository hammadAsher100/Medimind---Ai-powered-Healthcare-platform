================================================================================
                    HOW TO USE IN YOUR APPLICATION
================================================================================

import keras
import numpy as np
from PIL import Image

# Load model
model = keras.models.load_model('ml/registry/cnn_pneumonia_v3.h5')

# Predict
img = Image.open('xray.jpg').convert('RGB').resize((224, 224))
arr = np.array(img) / 255.0
arr = np.expand_dims(arr, axis=0)
prediction = model.predict(arr)[0][0]

if prediction >= 0.5:
    print(f"PNEUMONIA detected (confidence: {prediction:.2%})")
else:
    print(f"NORMAL - No pneumonia (confidence: {1-prediction:.2%})")
